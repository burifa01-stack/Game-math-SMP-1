Panduan Instalasi Firebase, Security Rules & Cloud Functions

Ikuti panduan praktis di bawah ini untuk menghubungkan MATH QUIZ ARENA — by Bu Rifa ke Firebase online agar perangkat guru dan HP siswa tersinkronisasi secara real-time dan kunci jawaban aman di server.

Langkah 1: Membuat Proyek Firebase (Gratis)

  1. Buka console.firebase.google.com dan masuk dengan akun Google Anda.
  2. Klik Tambah Proyek, beri nama math-quiz-arena-burifa.
  3. Aktifkan Authentication → Metode Masuk → aktifkan Email/Password, lalu tambahkan akun guru Anda di tab Users.
  4. Aktifkan Firestore Database → Create database → pilih lokasi terdekat (mis. asia-southeast2 Jakarta).
  5. Buka Project Settings (ikon gerigi) → Your apps → pilih ikon Web (</>) → salin objek firebaseConfig dan tempelkan pada tombol MODE DEMO / Konfigurasi Firebase di bagian atas aplikasi ini.

Langkah 2: Deploy ke Internet & Gunakan QR Code

  1. Simpan file ini sebagai index.html di dalam folder public/.
  2. Instal Firebase CLI di komputer: npm install -g firebase-tools
  3. Jalankan firebase login lalu firebase init hosting functions firestore.
  4. Jalankan firebase deploy. Anda akan mendapatkan tautan publik seperti https://math-quiz-arena-burifa.web.app.
  5. Buka tautan tersebut saat membuat sesi ulangan—QR Code otomatis mengarah ke URL publik tersebut sehingga siswa dapat langsung memindainya dari HP.
firestore.rules (Keamanan Database & Kunci Jawaban) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Dokumen soal publik (TANPA kunci jawaban) hanya bisa dibaca saat sesi aktif
    match /artifacts/{appId}/public/data/quiz_papers/{sessionCode} {
      allow read: if resource.data.status == 'active' || request.auth != null;
      allow write: if request.auth != null;
    }
    // Kunci jawaban HANYA boleh diakses oleh Guru terautentikasi atau Cloud Functions
    match /artifacts/{appId}/public/data/quiz_keys/{sessionCode} {
      allow read, write: if request.auth != null;
    }
    // Data pengerjaan siswa: siswa boleh membuat & update jawaban selama sesi aktif,
    // tetapi TIDAK boleh mengubah field 'score' atau 'gradedByServer'
    match /artifacts/{appId}/public/data/submissions/{submissionId} {
      allow read: if true;
      allow create: if true;
      allow update: if request.auth != null || (
        !request.resource.data.diff(resource.data).affectedKeys().hasAny(['score', 'gradedByServer'])
      );
      allow delete: if request.auth != null;
    }
  }
}
functions/index.js (Penilaian Aman di Server) 
const { onDocumentUpdated } = require("firebase-functions/v2/firestore");
const admin = require("firebase-admin");
admin.initializeApp();

exports.gradeStudentSubmission = onDocumentUpdated(
  "artifacts/{appId}/public/data/submissions/{submissionId}",
  async (event) => {
    const after = event.data.after.data();
    const before = event.data.before.data();
    // Jalankan penilaian hanya saat status berubah menjadi 'submitted'
    if (before.status === "submitted" || after.status !== "submitted") return null;

    const { appId } = event.params;
    const keySnap = await admin.firestore()
      .doc(`artifacts/${appId}/public/data/quiz_keys/${after.sessionCode}`)
      .get();

    if (!keySnap.exists) return null;
    const keyData = keySnap.data();
    let earnedPoints = 0;
    let maxPoints = 0;

    keyData.questions.forEach((q) => {
      maxPoints += Number(q.points || 10);
      const ans = after.answers?.[q.id];
      if (q.type === "pg" || q.type === "benar_salah") {
        if (String(ans) === String(q.correctAnswer)) earnedPoints += Number(q.points || 10);
      } else if (q.type === "isian") {
        if (String(ans || "").trim().toLowerCase() === String(q.correctAnswer).trim().toLowerCase()) {
          earnedPoints += Number(q.points || 10);
        }
      } else if (q.type === "pg_kompleks") {
        const expected = [...(q.correctAnswer || [])].sort().join(",");
        const given = [...(ans || [])].sort().join(",");
        if (expected === given && expected.length > 0) earnedPoints += Number(q.points || 10);
      }
    });

    const finalScore = maxPoints > 0 ? Math.round((earnedPoints / maxPoints) * 100) : 0;
    return event.data.after.ref.update({
      earnedPoints,
      maxPoints,
      score: finalScore,
      gradedByServer: true,
      gradedAt: admin.firestore.FieldValue.serverTimestamp(),
    });
  }
);
